# Template
앞에서 함수 템플릿, 클래스 템플릿의 템플릿 매개변수는 자료형이 들어갔었다.   
이번에는 자료형이 아닌 템플릿 매개변수를 사용해보자.   



```cpp
//MyArray.h
#include <assert.h>
#include <iostream>

template<typename T, unsigned int T_SIZE> // 이전 m_length를 대체하는 T_SIZE, 이렇게 하면 T_SIZE가 컴파일 타임에 결정됨.  
// 동적할당과 다름. T_SIZE뿐 아니라 T 또한 instantiation되기 위해서는 무조건 컴파일 시점에 결정에 결정.   
class MyArray
{
    private:
        T *m_data; // T m_data[T_SIZE]

    public:
        MyArray(int length)
        {
            m_data = new T[T_SIZE]; 
        }

        ~MyArray()
        {
            delete[] m_data;
        }

        T &operator[](int index) 
        {
            assert(index >= 0 && index < m_length);
            return m_data[index];
        }

        int getLength()
        {
            return T_SIZE;
        }

        void print()
        {
            for (int i = 0; i < m_length; i++)
            {
                std::cout << m_data[i] << " ";
            }
            std::cout << std::endl;
        }
};
```

```cpp
//main.cpp
#include "MyArray.h"

int main()
{
    MyArray<double, 100> my_array; //std::array<double, 100>

    for (int i = 0; i < my_array.get_Length(); i++)
    {
        my_array[i] = i + 65;
    }

    my_array.print();

    return 0;
}

```
